using System;

namespace PongGame
{
    // Класс для хранения координат
    public class GamePoint
    {
        public int X { get; set; }
        public int Y { get; set; }

        public GamePoint(int x, int y)
        {
            X = x;
            Y = y;
        }
    }

    // Класс для вектора скорости
    public class GameVector
    {
        public int X { get; set; }
        public int Y { get; set; }

        public GameVector(int x, int y)
        {
            X = x;
            Y = y;
        }
    }

    // Основной класс мяча
    public class PlayerBall
    {
        public GamePoint Position { get; set; }
        public GameVector Velocity { get; set; }

        public PlayerBall(GamePoint position, GameVector velocity)
        {
            Position = position;
            Velocity = velocity;
        }

        public void Move()
        {
            Position.X += Velocity.X;
            Position.Y += Velocity.Y;
        }

        public void BounceHorizontal()
        {
            Velocity.X *= -1;
        }

        public void BounceVertical()
        {
            Velocity.Y *= -1;
        }
    }

    // Класс ракетки
    public class PlayerPaddle
    {
        public GamePoint Position { get; set; }
        public int Height { get; } = 10;
        public int Width { get; } = 3;
        public int Speed { get; } = 5;

        public PlayerPaddle(GamePoint position)
        {
            Position = position;
        }

        public void MoveUp()
        {
            Position.Y = Math.Max(0, Position.Y - Speed);
        }

        public void MoveDown(int fieldHeight)
        {
            Position.Y = Math.Min(fieldHeight - Height, Position.Y + Speed);
        }
    }

    // Класс ИИ-ракетки
    public class AiPaddle : PlayerPaddle
    {
        public AiPaddle(GamePoint position) : base(position) { }

        public void FollowBall(PlayerBall ball, int fieldHeight)
        {
            var targetY = ball.Position.Y;
            if (Position.Y < targetY && Position.Y + Height < fieldHeight)
                MoveDown(fieldHeight);
            else if (Position.Y > targetY)
                MoveUp();
        }
    }

    // Счётчик очков
   public class Score
{
    public int PlayerScore { get; private set; }
    public int AiScore { get; private set; }

    public void AddPlayerScore()
    {
        PlayerScore++;
    }

    public void AddAiScore()
    {
        AiScore++;
    }

    public override string ToString()
    {
        return $"Игрок: {PlayerScore} | Компьютер: {AiScore}";
    }
}
    // Основной игровой объект
   public class Game
{
    private PlayerBall _ball;
    public PlayerBall Ball => _ball;
    public PlayerPaddle PlayerPaddle { get; }
    public AiPaddle AiPaddle { get; }
    public Score Score { get; } = new Score();

    private readonly int _fieldWidth = 80;
    private readonly int _fieldHeight = 20;

    public Game()
    {
        // Инициализация игровых объектов
        _ball = new PlayerBall(
            new GamePoint(_fieldWidth / 2, _fieldHeight / 2),
            new GameVector(2, 2));

        PlayerPaddle = new PlayerPaddle(
            new GamePoint(2, _fieldHeight / 2 - 5));

        AiPaddle = new AiPaddle(
            new GamePoint(_fieldWidth - 5, _fieldHeight / 2 - 5));
    }

    public void Update()
    {
            // Обновление позиции мяча
            Ball.Move();

            // Проверка столкновений
            CheckCollisions();

        // Обновление ИИ
        AiPaddle.FollowBall(Ball, _fieldHeight);
    }

    private void CheckCollisions()
    {
        // Столкновение с верхней/нижней границей
        if (Ball.Position.Y <= 0 || Ball.Position.Y >= _fieldHeight)
            Ball.BounceVertical();

        // Столкновение с ракетками
        if (IsBallCollidingWithPaddle(PlayerPaddle) ||
            IsBallCollidingWithPaddle(AiPaddle))
            Ball.BounceHorizontal();

        // Проверка на промах
        if (Ball.Position.X <= 0)
            Score.AddAiScore();
        else if (Ball.Position.X >= _fieldWidth)
                Score.AddPlayerScore();
    }

    private bool IsBallCollidingWithPaddle(PlayerPaddle paddle)
    {
        // Проверка столкновения с ракеткой
        return Ball.Position.X == paddle.Position.X + paddle.Width &&
               Ball.Position.Y >= paddle.Position.Y &&
               Ball.Position.Y <= paddle.Position.Y + paddle.Height;
    }
}
}
